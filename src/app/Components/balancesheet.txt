"use client";

import { useState, useEffect } from "react";
import { Line } from "react-chartjs-2";
import "chart.js/auto";
import "animate.css";

export default function BalanceSheet() {
  const [balanceData, setBalanceData] = useState({
    assets: 200000,
    liabilities: 80000,
    equity: 120000,
  });

  const [darkMode, setDarkMode] = useState(false);
  const [themeColor, setThemeColor] = useState("purple");

  useEffect(() => {
    const interval = setInterval(() => {
      setBalanceData((prev) => ({
        assets: prev.assets + Math.floor(Math.random() * 5000),
        liabilities: prev.liabilities + Math.floor(Math.random() * 3000),
        equity: prev.assets - prev.liabilities,
      }));
    }, 5000);

    return () => clearInterval(interval);
  }, []);

  const handleExport = () => {
    const data = JSON.stringify(balanceData, null, 2);
    const blob = new Blob([data], { type: "application/json" });
    const a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download = "balance-sheet.json";
    a.click();
  };

  const chartData = {
    labels: ["Q1", "Q2", "Q3", "Q4"],
    datasets: [
      {
        label: "Assets",
        data: [180000, 190000, 200000, balanceData.assets],
        borderColor: themeColor,
        backgroundColor: `rgba(${themeColor === "purple" ? "128,0,128" : "0,0,255"},0.2)`,
      },
      {
        label: "Liabilities",
        data: [70000, 75000, 80000, balanceData.liabilities],
        borderColor: "red",
        backgroundColor: "rgba(255,0,0,0.2)",
      },
      {
        label: "Equity",
        data: [110000, 115000, 120000, balanceData.equity],
        borderColor: "green",
        backgroundColor: "rgba(0,255,0,0.2)",
      },
    ],
  };

  return (
    <div className={`${darkMode ? "bg-gray-900 text-white" : "bg-gray-100 text-black"} min-h-screen flex flex-col items-center justify-center p-6`}>
      <div className="max-w-4xl w-full p-8 rounded-xl shadow-lg animate__animated animate__fadeInUp">
        <h1 className="text-3xl font-bold text-center">Balance Sheet Overview</h1>

        {/* Theme Controls */}
        <div className="flex justify-between mt-4">
          <button className="px-4 py-2 rounded bg-gray-700 text-white hover:bg-gray-600" onClick={() => setDarkMode(!darkMode)}>
            {darkMode ? "Light Mode" : "Dark Mode"}
          </button>
          <select className="px-4 py-2 rounded bg-gray-200 text-black" onChange={(e) => setThemeColor(e.target.value)}>
            <option value="purple">Purple Theme</option>
            <option value="blue">Blue Theme</option>
          </select>
        </div>

        {/* Financial Data */}
        <div className="grid grid-cols-3 gap-4 text-center mt-6 font-semibold">
          <div className="p-4 rounded-md shadow-md" style={{ backgroundColor: themeColor === "purple" ? "#D8BFD8" : "#87CEFA" }}>
            <h2>Assets</h2>
            <p>₦{balanceData.assets}</p>
          </div>
          <div className="bg-red-100 p-4 rounded-md shadow-md">
            <h2>Liabilities</h2>
            <p>₦{balanceData.liabilities}</p>
          </div>
          <div className="bg-green-100 p-4 rounded-md shadow-md">
            <h2>Equity</h2>
            <p>₦{balanceData.equity}</p>
          </div>
        </div>

        {/* Financial Trends */}
        <div className="mt-8">
          <h2 className="text-xl font-bold text-center">Financial Trends</h2>
          <Line data={chartData} />
        </div>

        {/* Export Functionality */}
        <button className="mt-6 bg-blue-500 text-white px-4 py-2 rounded shadow hover:bg-blue-600 transition" onClick={handleExport}>
          📁 Export Balance Sheet
        </button>
      </div>
    </div>
  );
}
